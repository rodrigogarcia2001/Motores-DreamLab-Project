using UnityEngine;

public class EnemyAttack : MonoBehaviour
{
    public float moveSpeed = 3f;         // Velocidad de movimiento
    public float attackRange = 1.5f;     // Distancia para atacar
    public int damage = 10;              // Daño que inflige
    public float attackCooldown = 2f;    // Tiempo entre ataques

    private float lastAttackTime = 0f;
    private Transform playerTransform;
    private Player playerScript;         // Referencia directa al script del jugador

    void Start()
    {
        // Buscar al jugador por tag
        GameObject playerObj = GameObject.FindWithTag("Player");

        if (playerObj != null)
        {
            playerTransform = playerObj.transform;
            playerScript = playerObj.GetComponent<Player>();

            if (playerScript == null)
            {
                Debug.LogError("El GameObject con tag 'Player' no tiene el script Player.");
            }
        }
        else
        {
            Debug.LogError("No se encontró ningún objeto con el tag 'Player'.");
        }
    }

    void Update()
    {
        if (playerTransform == null || playerScript == null)
            return;

        float distance = Vector3.Distance(transform.position, playerTransform.position);

        if (distance > attackRange)
        {
            // Mover hacia el jugador
            transform.position = Vector3.MoveTowards(
                transform.position,
                playerTransform.position,
                moveSpeed * Time.deltaTime
            );

            // Hacer que mire al jugador (opcional)
            Vector3 lookDirection = playerTransform.position - transform.position;
            lookDirection.y = 0; // Evita que rote en el eje Y si es un juego 2D o de top-down
            if (lookDirection != Vector3.zero)
                transform.rotation = Quaternion.LookRotation(lookDirection);
        }
        else
        {
            // Atacar si pasó suficiente tiempo
            if (Time.time >= lastAttackTime + attackCooldown)
            {
                playerScript.TakeDamage(damage);
                lastAttackTime = Time.time;
            }
        }
    }
}
